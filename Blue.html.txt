<script>
let settings = { whatsapp:'966500000000', badge:'وصول جديد 2025', title:'وصلت المنتجات الجديدة!', subtitle:'اختر الكميات المطلوبة وأكمل طلبك عبر الواتساب' };
let products = [{"id":1,"image":"🧴","name":"كريم ترطيب","desc":"ترطيب فائق","price":50,"unit":"كرتون","badge":"جديد"}];
let quantities = {};
const ADMIN_PASSWORD = 'bluesky2025';

function loadData(){
    const s = localStorage.getItem('bs_settings'); if(s) settings = JSON.parse(s);
    const p = localStorage.getItem('bs_products'); if(p) products = JSON.parse(p);
}

function renderCatalog(filter = ''){
    const grid = document.getElementById('productsGrid');
    const filtered = products.filter(p => p.name.includes(filter));
    grid.innerHTML = filtered.map(p => `
        <div class="product-card">
            <div class="product-img-wrap">${p.image.length > 5 ? `<img src="${p.image}">` : `<span style="font-size:80px">${p.image}</span>`}</div>
            <div class="product-info">
                <div class="product-name">${p.name}</div>
                <div style="display:flex; justify-content:space-between; align-items:center; margin-top:15px;">
                    <div style="font-weight:bold;">$${p.price}</div>
                    <div class="qty-ctrl">
                        <button class="qty-btn" onclick="updateQty(${p.id}, -1)">-</button>
                        <span id="qty-${p.id}">${quantities[p.id] || 0}</span>
                        <button class="qty-btn" onclick="updateQty(${p.id}, 1)">+</button>
                    </div>
                </div>
            </div>
        </div>
    `).join('');
}

function filterProducts(){ renderCatalog(document.getElementById('searchInput').value); }

function updateQty(id, delta){
    quantities[id] = Math.max(0, (quantities[id] || 0) + delta);
    document.getElementById(`qty-${id}`).textContent = quantities[id];
    updateSummary();
}

function updateSummary(){
    const orderItems = document.getElementById('orderItems');
    const selected = products.filter(p => quantities[p.id] > 0);
    let total = 0; let count = 0;
    if(selected.length === 0){
        orderItems.innerHTML = 'السلة فارغة';
        document.getElementById('totalPrice').textContent = '$0';
        document.getElementById('floatingCart').style.display = 'none';
        return;
    }
    orderItems.innerHTML = selected.map(p => {
        const sub = p.price * quantities[p.id]; total += sub; count += quantities[p.id];
        return `<div style="display:flex; justify-content:space-between;"><span>${p.name} × ${quantities[p.id]}</span><span>$${sub}</span></div>`;
    }).join('');
    document.getElementById('totalPrice').textContent = `$${total}`;
    document.getElementById('floatingCart').style.display = 'flex';
    document.getElementById('cartCount').textContent = `${selected.length} أصناف`;
    document.getElementById('cartTotal').textContent = `الإجمالي: $${total}`;
}

function toggleAdmin(){ document.getElementById('pwOverlay').classList.add('show'); }
function checkPassword(){
    if(document.getElementById('pwInput').value === ADMIN_PASSWORD){
        document.getElementById('pwOverlay').classList.remove('show');
        document.getElementById('adminPanel').classList.add('open');
        renderAdmin();
    } else { document.getElementById('pwErr').textContent = 'خطأ!'; }
}
function closePwDialog(){ document.getElementById('pwOverlay').classList.remove('show'); }

function renderAdmin(){
    document.getElementById('set-whatsapp').value = settings.whatsapp;
    document.getElementById('set-title').value = settings.title;
    document.getElementById('adminProductsGrid').innerHTML = products.map(p => `
        <div class="admin-card">
            <input type="text" value="${p.name}" onchange="updateProd(${p.id}, 'name', this.value)">
            <input type="number" value="${p.price}" onchange="updateProd(${p.id}, 'price', this.value)">
            <button onclick="removeProd(${p.id})">حذف</button>
        </div>
    `).join('');
}
function updateProd(id, k, v){ const p = products.find(x=>x.id===id); if(p) p[k] = (k==='price'?parseFloat(v):v); }
function addProduct(){ products.push({id:Date.now(), name:'جديد', price:0, image:'📦'}); renderAdmin(); }
function removeProd(id){ products = products.filter(x=>x.id!==id); renderAdmin(); }

function saveAndClose(){
    settings.whatsapp = document.getElementById('set-whatsapp').value;
    settings.title = document.getElementById('set-title').value;
    localStorage.setItem('bs_settings', JSON.stringify(settings));
    localStorage.setItem('bs_products', JSON.stringify(products));
    location.reload();
}

function submitOrder(){
    const name = document.getElementById('clientName').value;
    const selected = products.filter(p => quantities[p.id] > 0);
    if(!name || selected.length === 0) { showToast('أكمل البيانات'); return; }
    let msg = `طلب من: ${name}\n`;
    selected.forEach(p => msg += `• ${p.name} × ${quantities[p.id]}\n`);
    window.open(`https://wa.me/${settings.whatsapp}?text=${encodeURIComponent(msg)}`, '_blank');
}

function closeModal(){ document.getElementById('successModal').classList.remove('show'); }
function showToast(m){ const t = document.getElementById('toast'); t.textContent = m; t.classList.add('show'); setTimeout(()=>t.classList.remove('show'), 2000); }

loadData();
renderCatalog();
document.getElementById('catalogTitle').textContent = settings.title;
</script>
